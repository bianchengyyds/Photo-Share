### 表 1：album（相册表）

```sql
CREATE TABLE album (
    id          VARCHAR(36)  NOT NULL COMMENT '相册ID（UUID）',
    title       VARCHAR(100) NOT NULL COMMENT '相册标题',
    description VARCHAR(500) DEFAULT NULL COMMENT '相册描述',
    cover_url   VARCHAR(500) DEFAULT NULL COMMENT '封面照片URL',
    created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    updated_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='相册表';

-- 相册表添加软删除字段
ALTER TABLE album ADD COLUMN is_deleted TINYINT(1) NOT NULL DEFAULT 0 COMMENT '是否删除（0=正常，1=已删除）';
```

| 字段        | 类型         | 说明               |
| ----------- | ------------ | ------------------ |
| id          | VARCHAR(36)  | UUID 主键          |
| title       | VARCHAR(100) | 相册标题，必填     |
| description | VARCHAR(500) | 相册描述，可选     |
| cover_url   | VARCHAR(500) | 封面图 URL         |
| created_at  | DATETIME     | 创建时间，自动生成 |
| updated_at  | DATETIME     | 更新时间，自动更新 |

### 表 2：photo（照片表）

```sql
CREATE TABLE photo (
    id          VARCHAR(36)  NOT NULL COMMENT '照片ID（UUID）',
    album_id    VARCHAR(36)  NOT NULL COMMENT '所属相册ID',
    file_name   VARCHAR(255) NOT NULL COMMENT '原始文件名',
    file_url    VARCHAR(500) NOT NULL COMMENT '文件存储路径/URL',
    file_size   BIGINT       NOT NULL COMMENT '文件大小（字节）',
    mime_type   VARCHAR(50)  NOT NULL COMMENT '文件类型（image/jpeg等）',
    sort_order  INT          NOT NULL DEFAULT 0 COMMENT '排序序号',
    created_at  DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '上传时间',
    PRIMARY KEY (id),
    INDEX idx_album_id (album_id),
    CONSTRAINT fk_photo_album FOREIGN KEY (album_id) REFERENCES album(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='照片表';


-- 照片表添加软删除字段
ALTER TABLE photo ADD COLUMN is_deleted TINYINT(1) NOT NULL DEFAULT 0 COMMENT '是否删除（0=正常，1=已删除）';
```

| 字段       | 类型         | 说明                                   |
| ---------- | ------------ | -------------------------------------- |
| id         | VARCHAR(36)  | UUID 主键                              |
| album_id   | VARCHAR(36)  | 外键，关联相册                         |
| file_name  | VARCHAR(255) | 原始文件名（如 `photo.jpg`）           |
| file_url   | VARCHAR(500) | 存储路径（本地路径或 OSS URL）         |
| file_size  | BIGINT       | 文件大小，单位字节                     |
| mime_type  | VARCHAR(50)  | MIME 类型（`image/jpeg`、`image/png`） |
| sort_order | INT          | 排序序号，默认 0                       |
| created_at | DATETIME     | 上传时间                               |

**索引说明**：
- `idx_album_id`：按相册查询照片时加速
- `ON DELETE CASCADE`：删除相册时自动删除其下所有照片

### 表 3：share_link（分享链接表）

```sql
CREATE TABLE share_link (
    id             VARCHAR(36)  NOT NULL COMMENT '分享链接ID（UUID）',
    album_id       VARCHAR(36)  NOT NULL COMMENT '关联相册ID',
    share_code     VARCHAR(20)  NOT NULL COMMENT '分享码（8位短码）',
    password       VARCHAR(100) DEFAULT NULL COMMENT '访问密码（加密存储）',
    expire_at      DATETIME     DEFAULT NULL COMMENT '过期时间（NULL表示永久）',
    allow_download TINYINT(1)   NOT NULL DEFAULT 1 COMMENT '是否允许下载（1=允许，0=禁止）',
    visit_count    INT          NOT NULL DEFAULT 0 COMMENT '访问次数',
    created_at     DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '创建时间',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_share_code (share_code),
    INDEX idx_album_id (album_id),
    CONSTRAINT fk_share_album FOREIGN KEY (album_id) REFERENCES album(id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='分享链接表';
```

| 字段           | 类型         | 说明                              |
| -------------- | ------------ | --------------------------------- |
| id             | VARCHAR(36)  | UUID 主键                         |
| album_id       | VARCHAR(36)  | 外键，关联相册                    |
| share_code     | VARCHAR(20)  | 分享码（如 `a3Bx9Kz2`），唯一     |
| password       | VARCHAR(100) | 访问密码，BCrypt 加密存储，可为空 |
| expire_at      | DATETIME     | 过期时间，NULL = 永久有效         |
| allow_download | TINYINT(1)   | 1 = 允许下载，0 = 禁止下载        |
| visit_count    | INT          | 累计访问次数                      |
| created_at     | DATETIME     | 创建时间                          |

**索引说明**：
- `uk_share_code`：分享码唯一索引，通过分享码快速查找
- `idx_album_id`：查询某个相册的所有分享链接

-- ============================================
-- 点赞、评论、标签功能数据库迁移脚本
-- 执行前请备份现有数据

-- 1. 点赞表：相册和照片共用一张表，用 target_type 区分
--    （like 是 MySQL 关键字，因此表名为 user_like）
CREATE TABLE user_like (
    id          VARCHAR(36) NOT NULL COMMENT '点赞记录ID（UUID）',
    user_id     VARCHAR(36) NOT NULL COMMENT '点赞人ID',
    target_type VARCHAR(10) NOT NULL COMMENT '目标类型：ALBUM=相册，PHOTO=照片',
    target_id   VARCHAR(36) NOT NULL COMMENT '目标ID（相册ID或照片ID）',
    created_at  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '点赞时间',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_user_target (user_id, target_type, target_id),
    INDEX idx_target (target_type, target_id),
    INDEX idx_user_type (user_id, target_type, created_at),
    CONSTRAINT fk_like_user FOREIGN KEY (user_id) REFERENCES `user` (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='点赞表（相册/照片共用）';

-- 2. 相册评论表：只有相册可以被评论，照片不提供评论
CREATE TABLE album_comment (
    id         VARCHAR(36)  NOT NULL COMMENT '评论ID（UUID）',
    album_id   VARCHAR(36)  NOT NULL COMMENT '被评论的相册ID',
    user_id    VARCHAR(36)  NOT NULL COMMENT '评论人ID',
    content    VARCHAR(500) NOT NULL COMMENT '评论内容',
    is_deleted TINYINT(1)   NOT NULL DEFAULT 0 COMMENT '是否删除（0=正常，1=已删除）',
    created_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '评论时间',
    updated_at DATETIME     NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP COMMENT '更新时间',
    PRIMARY KEY (id),
    INDEX idx_album_time (album_id, created_at),
    INDEX idx_user_id (user_id),
    CONSTRAINT fk_comment_album FOREIGN KEY (album_id) REFERENCES album (id) ON DELETE CASCADE,
    CONSTRAINT fk_comment_user FOREIGN KEY (user_id) REFERENCES `user` (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='相册评论表';

-- 3. 标签字典表：同名标签只存一行，便于按标签聚合与筛选
CREATE TABLE tag (
    id         VARCHAR(36) NOT NULL COMMENT '标签ID（UUID）',
    name       VARCHAR(50) NOT NULL COMMENT '标签名（首尾空格已去除，不区分大小写唯一）',
    created_at DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '首次被使用时间',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='标签字典表';

-- 4. 标签关联表：一个标签打在一次相册或一张照片上，并记录添加人
--    任何登录用户都能添加标签，但只能删除自己添加的那一条
CREATE TABLE tag_relation (
    id          VARCHAR(36) NOT NULL COMMENT '关联ID（UUID）',
    tag_id      VARCHAR(36) NOT NULL COMMENT '标签ID',
    user_id     VARCHAR(36) NOT NULL COMMENT '添加人ID',
    target_type VARCHAR(10) NOT NULL COMMENT '目标类型：ALBUM=相册，PHOTO=照片',
    target_id   VARCHAR(36) NOT NULL COMMENT '目标ID（相册ID或照片ID）',
    created_at  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '添加时间',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_tag_target (tag_id, target_type, target_id, user_id),
    INDEX idx_target (target_type, target_id),
    INDEX idx_user_id (user_id),
    CONSTRAINT fk_tagrel_tag FOREIGN KEY (tag_id) REFERENCES tag (id) ON DELETE CASCADE,
    CONSTRAINT fk_tagrel_user FOREIGN KEY (user_id) REFERENCES `user` (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='标签关联表（相册/照片共用）';

-- ============================================
-- 迁移完成
-- 说明：点赞数、评论数不落库冗余，一律实时 COUNT，避免与明细不一致；
--       target_id 因为要同时指向相册和照片（多态），无法建外键约束，
--       由查询侧 JOIN album / photo 并过滤 is_deleted 来屏蔽已删除对象。

-- 1. 相册增加可见范围：默认私密，仅本人与分享链接可访问；

-- 2. 关注关系表：单向关注，A 关注 B 不代表 B 关注 A（互关即两行）
CREATE TABLE user_follow (
    id          VARCHAR(36) NOT NULL COMMENT '关注记录ID（UUID）',
    follower_id VARCHAR(36) NOT NULL COMMENT '发起关注的人',
    followee_id VARCHAR(36) NOT NULL COMMENT '被关注的人',
    created_at  DATETIME    NOT NULL DEFAULT CURRENT_TIMESTAMP COMMENT '关注时间',
    PRIMARY KEY (id),
    UNIQUE INDEX uk_follow (follower_id, followee_id),
    INDEX idx_followee_time (followee_id, created_at),
    INDEX idx_follower_time (follower_id, created_at),
    CONSTRAINT fk_follow_follower FOREIGN KEY (follower_id) REFERENCES `user` (id) ON DELETE CASCADE,
    CONSTRAINT fk_follow_followee FOREIGN KEY (followee_id) REFERENCES `user` (id) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='用户关注表';

-- 3. 评论表支持一级回复：parent_id 指向被回复的那条评论，回复本身不再向下嵌套
ALTER TABLE album_comment
    ADD COLUMN parent_id        VARCHAR(36) DEFAULT NULL COMMENT '被回复的评论ID（NULL=一级评论）' AFTER album_id,
    ADD COLUMN reply_to_user_id VARCHAR(36) DEFAULT NULL COMMENT '被回复人ID，用于显示"回复 @某人"' AFTER parent_id,
    ADD INDEX idx_parent (parent_id);

-- 3.1 硬删除一级评论时连带清理其回复；被回复人注销后仍可展示历史回复
ALTER TABLE album_comment
    ADD CONSTRAINT fk_comment_parent FOREIGN KEY (parent_id) REFERENCES album_comment (id) ON DELETE CASCADE,
    ADD CONSTRAINT fk_comment_reply_user FOREIGN KEY (reply_to_user_id) REFERENCES `user` (id) ON DELETE SET NULL;

-- 4. 站内通知表：点赞、评论、回复、关注、@提及统一进这一张表
CREATE TABLE notification (
    id          VARCHAR(36)  NOT NULL COMMENT '通知ID（UUID）',
    user_id     VARCHAR(36)  NOT NULL COMMENT '接收人ID',
    actor_id    VARCHAR(36)  DEFAULT NULL COMMENT '触发人ID（NULL=系统或用户已注销）',
    type        VARCHAR(20)  NOT NULL COMMENT 'LIKE=点赞，COMMENT=评论，REPLY=回复，FOLLOW=关注，MENTION=@提及',
    target_type VARCHAR(20)  NOT NULL COMMENT '跳转目标类型：ALBUM=相册，PHOTO=照片，USER=用户（关注）',

## 设计要点

| 设计决策                         | 原因                                   |
| -------------------------------- | -------------------------------------- |
| **主键用 UUID**                  | 避免自增 ID 暴露数据量，分享链接更安全 |
| **share_code 唯一索引**          | 通过短码快速查找分享链接               |
| **ON DELETE CASCADE**            | 删除相册时自动清理照片和分享链接       |
| **password 用 VARCHAR(100)**     | 预留 BCrypt 加密后的长度（60字符）     |
| **expire_at 允许 NULL**          | NULL 表示永久有效，简化逻辑            |
| **allow_download 用 TINYINT(1)** | MySQL 的布尔类型实现                   |
| **utf8mb4 字符集**               | 支持 emoji 和特殊字符                  |
| **sort_order 字段**              | 预留照片排序功能                       |
| **点赞表名为 user_like**         | `like` 是 MySQL 关键字，不能直接做表名  |
| **target_type + target_id 多态** | 一张表同时支持相册和照片点赞、标签，避免四张近似表 |
| **多态 target_id 不建外键**      | 外键只能指向单表；改由查询侧 JOIN 并过滤 is_deleted |
| **标签拆成 tag + tag_relation**  | 同名标签只存一行，便于聚合、改名和统计   |
| **关注关系即授权**               | 搜索和动态流只覆盖「本人 + 我关注的人」，未关注者仍只能靠分享链接访问相册 |
| **album.visibility 默认 PRIVATE**| 升级不会把存量相册意外公开，需用户在编辑相册时主动放开给关注者 |
| **回复只做一级（parent_id）**    | 两级足够表达对话，楼中楼的递归查询与渲染成本高 |
| **冗余存 reply_to_user_id**      | 列表直接显示「回复 @某人」并给被回复人发通知，省掉一次父查询 |
| **通知一张表装五类事件**         | type + target 多态即可承载跳转，避免点赞/评论/回复/关注/@ 五张近似表 |
| **actor_id 用 ON DELETE SET NULL** | 注销用户不该带走别人的历史通知，展示时降级为「用户已注销」 |
| **@提及不建 mention 表**         | 正文里的 @昵称 只做展示，真实关系落在 notification，少一张表和一次 JOIN |
| **搜索用 LIKE 前后通配**         | 数据量小，暂不引入 FULLTEXT/ngram 中文分词；先靠归属+可见性索引收窄范围 |
